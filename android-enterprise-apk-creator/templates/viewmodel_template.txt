package com.[company].[appname].presentation.[feature_name]

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.[company].[appname].domain.model.[ENTITY_NAME]
import com.[company].[appname].domain.usecase.[ENTITY_NAME]UseCases
import dagger.hilt.android.lifecycle.HiltViewModel
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.catch
import kotlinx.coroutines.flow.launchIn
import kotlinx.coroutines.flow.onEach
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch
import javax.inject.Inject

/**
 * ViewModel for [FEATURE_NAME] screen
 */
@HiltViewModel
class [FEATURE_NAME]ViewModel @Inject constructor(
    private val [entity_name_lower]UseCases: [ENTITY_NAME]UseCases
) : ViewModel() {
    private val _state = MutableStateFlow([FEATURE_NAME]State())
    val state: StateFlow<[FEATURE_NAME]State> = _state.asStateFlow()
    
    init {
        loadData()
    }
    
    private fun loadData() {
        viewModelScope.launch {
            _state.update { it.copy(isLoading = true) }
            try {
                [entity_name_lower]UseCases.getAll()
                    .catch { e ->
                        _state.update { 
                            it.copy(isLoading = false, error = e.message ?: "Unknown error")
                        }
                    }
                    .collect { items ->
                        _state.update { 
                            it.copy([entity_name_lower] = items, isLoading = false, error = null)
                        }
                    }
            } catch (e: Exception) {
                _state.update { 
                    it.copy(isLoading = false, error = e.message ?: "Failed to load data")
                }
            }
        }
    }
    
    fun refresh() {
        loadData()
    }
    
    fun add[ENTITY_NAME](entity: [ENTITY_NAME]) {
        viewModelScope.launch {
            _state.update { it.copy(isLoading = true) }
            try {
                [entity_name_lower]UseCases.insert(entity)
                loadData()
            } catch (e: Exception) {
                _state.update { 
                    it.copy(isLoading = false, error = "Failed to add: ${e.message}")
                }
            }
        }
    }
    
    fun update[ENTITY_NAME](entity: [ENTITY_NAME]) {
        viewModelScope.launch {
            _state.update { it.copy(isLoading = true) }
            try {
                [entity_name_lower]UseCases.update(entity)
                loadData()
            } catch (e: Exception) {
                _state.update { 
                    it.copy(isLoading = false, error = "Failed to update: ${e.message}")
                }
            }
        }
    }
    
    fun delete[ENTITY_NAME](entity: [ENTITY_NAME]) {
        viewModelScope.launch {
            _state.update { it.copy(isLoading = true) }
            try {
                [entity_name_lower]UseCases.delete(entity)
                loadData()
            } catch (e: Exception) {
                _state.update { 
                    it.copy(isLoading = false, error = "Failed to delete: ${e.message}")
                }
            }
        }
    }
}

data class [FEATURE_NAME]State(
    val [entity_name_lower]: List<[ENTITY_NAME]> = emptyList(),
    val isLoading: Boolean = false,
    val error: String? = null,
    val selected[ENTITY_NAME]: [ENTITY_NAME]? = null
)
